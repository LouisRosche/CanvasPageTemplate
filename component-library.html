Pomodoro Timer

⏱️ Focus Timer

25:00
<div style="background: white; border: 2px solid #667EEA; padding: 20px; border-radius: 8px; text-align: center; max-width: 300px;">
    <h4>⏱️ Focus Timer</h4>
    <div id="timer" style="font-size: 3em; font-weight: 700; color: #667EEA;">25:00</div>
    <button onclick="startTimer()">Start</button>
    <button onclick="pauseTimer()">Pause</button>
    <button onclick="resetTimer()">Reset</button>
</div>

<script>
let timeLeft = 1500; // 25 minutes in seconds
let timerInterval;

function startTimer() {
    timerInterval = setInterval(() => {
        if (timeLeft > 0) {
            timeLeft--;
            updateDisplay();
        } else {
            clearInterval(timerInterval);
            alert('Time\'s up! Great work!');
        }
    }, 1000);
}

function pauseTimer() {
    clearInterval(timerInterval);
}

function resetTimer() {
    clearInterval(timerInterval);
    timeLeft = 1500;
    updateDisplay();
}

function updateDisplay() {
    const minutes = Math.floor(timeLeft / 60);
    const seconds = timeLeft % 60;
    document.getElementById('timer').textContent =
        `${minutes}:${seconds.toString().padStart(2, '0')}`;
}
</script>

        <li><input type="checkbox" onchange="updateProgress()"> Task 1</li>
        <li><input type="checkbox" onchange="updateProgress()"> Task 2</li>
    </ul>

<script>
function updateProgress() {
    const checkboxes = document.querySelectorAll('input[type="checkbox"]');
    const checked = Array.from(checkboxes).filter(cb => cb.checked).length;
    const percentage = Math.round((checked / checkboxes.length) * 100);
    document.getElementById('progressBar').style.width = percentage + '%';
    document.getElementById('progressBar').textContent = percentage + '%';
}
</script>
